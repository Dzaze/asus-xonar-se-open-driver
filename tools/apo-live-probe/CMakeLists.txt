cmake_minimum_required(VERSION 3.25)
project(XonarApoLiveProbe LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Cette sonde exige Windows et MSVC.")
endif()
add_executable(apo-live-probe main.cpp)
target_compile_features(apo-live-probe PRIVATE cxx_std_20)
target_compile_definitions(apo-live-probe PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(apo-live-probe PRIVATE ole32 uuid advapi32 cfgmgr32 AudioBaseProcessingObjectV140)
enable_testing()
set(XONAR_APO_TEST_DLL "" CACHE FILEPATH "DLL explicite à tester hors matériel (configuration identité)")
if(XONAR_APO_TEST_DLL)
    add_test(NAME apo.registration-format-routing COMMAND apo-live-probe --inspect "${XONAR_APO_TEST_DLL}")
endif()
